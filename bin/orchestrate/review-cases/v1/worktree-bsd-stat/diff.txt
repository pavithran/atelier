+  const dir = mkdtempSync(join(tmpdir(), "atelier-run-agent-worktree-"));
+  t.after(() => rmSync(dir, { recursive: true, force: true }));
+  const home = join(dir, "home"), main = join(dir, "main"), record = join(dir, "record.json");
+  mkdirSync(join(home, ".local", "bin"), { recursive: true });
+  standIn(join(home, ".local", "bin", "opencode-glm"), record);
+
+  // Create a repo with a commit, so we can make a worktree from it.
+  const gitMain = gitRepo(main);
+  writeFileSync(join(main, "file.txt"), "content");
+  gitMain("add", ".");
+  gitMain("commit", "--quiet", "-m", "initial");
+  gitMain("branch", "worktree-branch");
+
+  // Add a linked worktree.
+  const ws = join(dir, "worktree");
+  gitMain("worktree", "add", "--quiet", ws, "worktree-branch");
+
+  // Verify it is a worktree: .git is a file, not a directory.
+  assert.ok(existsSync(join(ws, ".git")), ".git exists");
+  const gitStat = execFileSync("stat", ["-f", "%HT", join(ws, ".git")], { encoding: "utf8" }).trim();
+  assert.equal(gitStat, "Regular File", ".git is a file in a worktree");
+

+  const result = spawnSync(zsh, [join(repo, "bin", "orchestrate", "run-agent.sh"), "glm", ws, join(dir, "out.txt"), "A prompt."], {

+  });
+  assert.equal(result.status, 0, result.stderr);
+

+  const seen = JSON.parse(readFileSync(record, "utf8"));

+

+  assert.equal(execFileSync("git", ["status", "--porcelain"], { cwd: ws, encoding: "utf8" }), "", ".scratch/ is kept out of Git in the worktree");

+  const excludePath = execFileSync("git", ["rev-parse", "--git-path", "info/exclude"], { cwd: ws, encoding: "utf8" }).trim();
